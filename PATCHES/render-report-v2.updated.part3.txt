            <th>How to verify</th>
            <th>Why this rank</th>
          </tr>
        </thead>
        <tbody>${rows}</tbody>
      </table>
    </div>
    ${secondaryCount

function conversionPathSection(model) {
  const paths = Array.isArray(model.conversionPaths) ? model.conversionPaths : [];
  const assessed = paths.filter((p) => p.status !== "Not Assessed");
  const clear = assessed.filter((p) => p.status === "Clear");
  const weak = assessed.filter((p) => p.status !== "Clear");

  const directAnswer = paths.length === 0 || assessed.length === 0
    ? "PRYSM could not determine whether visitors can move cleanly from interest to action because conversion-path evidence was unavailable."
    : weak.length === 0
      ? `${clear.length} assessed conversion path${clear.length === 1 ? " is" : "s are"} clear from entry through action.`
      : `${weak.length} of ${assessed.length} assessed conversion path${assessed.length === 1 ? "" : "s"} contains friction that can weaken confidence or interrupt movement toward action.`;

  const pathRows = paths.map((p) => {
    const status = p.status === "Clear"
      ? "PASS"
      : p.status === "Not Assessed"
        ? "UNAVAILABLE"
        : "FINDING";
    const explanation = p.status === "Clear"
      ? "The visible route contains a usable sequence toward action."
      : p.status === "Not Assessed"
        ? "The path could not be assessed from the available evidence."
        : (p.blockers || []).join("; ") || "The visible route contains friction before conversion.";
    return `<tr>
      <td><strong>${e(p.name || "Conversion path")}</strong></td>
      <td>${e((p.steps || []).join(" → ") || "Unavailable")}</td>
      <td><span class="chip ${status === "PASS" ? "cap-ok" : status === "FINDING" ? "cap-missing" : "cap-neutral"}">${e(status)}</span></td>
      <td class="small">${e(explanation)}</td>
    </tr>`;
  }).join("");

  const blockers = [...new Set(paths.flatMap((p) => p.blockers || []))];

  return `<section id="paths" class="card">
    <p style="font-size:1.15rem;font-weight:700;margin-bottom:6px">Can visitors easily move from interest to action?</p>
    <p class="muted small">Conversion Path Architecture</p>
    <h2>Conversion path architecture</h2>
    <p>${e(directAnswer)}</p>

    <h3>Primary conversion sequence</h3>
    <div style="overflow-x:auto;margin:16px 0">
      <svg viewBox="0 0 820 120" role="img" aria-label="Entry to service understanding to trust to primary CTA to conversion destination" style="width:100%;min-width:700px">
        <defs>
          <marker id="pathArrow" markerWidth="8" markerHeight="8" refX="7" refY="3" orient="auto">
            <path d="M0,0 L0,6 L7,3 z" fill="currentColor"/>
          </marker>
        </defs>
        ${["Entry","Service understanding","Trust / proof","Primary CTA","Conversion destination"].map((label, index) => {
          const x = 10 + index * 162;
          const width = 140;
          const line = index < 4 ? `<line x1="${x + width}" y1="58" x2="${x + 158}" y2="58" stroke="currentColor" marker-end="url(#pathArrow)" opacity=".45"/>` : "";
          return `<rect x="${x}" y="30" width="${width}" height="56" rx="10" fill="none" stroke="currentColor" opacity=".45"/>
            <text x="${x + width / 2}" y="62" text-anchor="middle" font-size="12">${e(label)}</text>${line}`;
        }).join("")}
      </svg>
    </div>

    <h3>Important path status</h3>
    ${pathRows
      ? `<div class="table-wrap"><table>
          <thead><tr><th>Path</th><th>Observed route</th><th>Status</th><th>Explanation</th></tr></thead>
          <tbody>${pathRows}</tbody>
        </table></div>`
      : `<p><span class="chip cap-neutral">UNAVAILABLE</span> No conversion-path evidence was available.</p>`}

    <h3>Where visitors may hesitate</h3>
    ${blockers.length
      ? `<ul>${blockers.map((b) => `<li><strong>${e(b)}</strong> — this weakens confidence or clarity before the action is completed.</li>`).join("")}</ul>`
      : `<p>No material hesitation point was identified in the assessed path evidence.</p>`}

    <h3>What is working</h3>
    ${clear.length
      ? `<ul>${clear.map((p) => `<li><strong>${e(p.name || "Conversion path")}:</strong> the assessed route is clear.</li>`).join("")}</ul>`
      : `<p>No assessed conversion path was strong enough to present as a clear strength.</p>`}

    <h3>Material findings only</h3>
    ${weak.length
      ? `<p>${e(weak.length)} assessed path${weak.length === 1 ? " requires" : "s require"} attention. Supporting blockers are shown above.</p>`
      : `<p>No material conversion-path finding was established.</p>`}

    <h3>Limitations</h3>
    ${paths.some((p) => p.status === "Not Assessed")
      ? `<div class="note"><strong>PARTIAL:</strong> At least one path could not be assessed. Missing path evidence was not treated as a failure.</div>`
      : `<p>The visible conversion paths shown above were assessable from the available model evidence.</p>`}
  </section>`;
}
